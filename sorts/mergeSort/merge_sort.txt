// A -> Array<inteiro>
// p -> indice da posição mais à esquerda considerada
// r -> indice da posição mais à direita considerada
procedimento mergesort(A, p, r){
    se p < r{
        q = floor((p+r) / 2); //floor é uma função de arredondamento para baixo, q é o elemento no meio do vetor
        mergesort(A, p, q);
        mergesort(A, q + 1, r);
        merge(A, p, q, r);
    }
}
procedimento merge(A, p, q, r){
    n1 = q - p + 1;  //quantidade de elementos na metade da esquerda
    n2 = r - q; //quantidade de elementos na metade da direita
    L = Array<inteiro>[n1 + 1];
    R = Array<inteiro>[n2 + 1];
    para i = 1 até n1{ //copia a metade esquerda de A para L
        L[i] = A[p + i - 1];
    }
    para j = 1 até n2{ //copia metade direita de A para L
        R[j] = A[q + j];
    }
    L[n1 + 1] = inf; //inf é o valor infinito positivo
    R[n2 + 1] = inf;
    i = 1;
    j = 1;
    para k = p até r{
        se L[i] <= R[j]{
            A[k] = L[i];
            i = i + 1;
        }
        senão{
            A[k] = R[j];
            j = j + 1;
        }
    }
}
