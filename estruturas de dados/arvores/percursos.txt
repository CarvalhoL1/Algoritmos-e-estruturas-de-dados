A ideia é percursos iterativos em pos/em/pos ordem, sem estruturas de dados auxiliares e nem recursão

estruturas basicas para os procedimentos:

Registro No{
    inteiro chave;
    No pai;
    No esq;
    No dir;
}
Registro Arvore{
    No raiz;
}

procedimento percursoPreOrdem (Arvore arv){
    ant = NIL;
    atual = arv.raiz;
    enquanto atual != NIL{
        se ant == atual.pai{
            imprimir(atual + " ");
            se atual.esq != NIL{
                prox = atual.esq;
            }
            senão se atual.dir != NIL{
                prox = atual.dir;
            }
            senão{
                prox = atual.pai;
            }
        }
        senão se ant == atual.esq{
            se atual.dir != NIL{
                prox = atual.dir;
            }
            senão{
                prox = atual.pai;
            }
        }
        senão{
            prox = atual.pai;
        }
        ant = atual;
        atual = prox;
    }
}
