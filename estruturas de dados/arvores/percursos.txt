A ideia é percursos iterativos em pos/em/pos ordem (para arvores binarias de busca), sem estruturas de dados auxiliares e nem recursão

estruturas basicas para os procedimentos:

Registro No{
    inteiro chave;
    No pai;
    No esq;
    No dir;
}
Registro Arvore{
    No raiz;
}

procedimento percursoPreOrdem (Arvore arv){
    ant = NIL;
    atual = arv.raiz;
    enquanto atual != NIL{
        se ant == atual.pai{
            imprimir(atual.chave + " ");
            se atual.esq != NIL{
                prox = atual.esq;
            }
            senão se atual.dir != NIL{
                prox = atual.dir;
            }
            senão{
                prox = atual.pai;
            }
        }
        senão se ant == atual.esq{
            se atual.dir != NIL{
                prox = atual.dir;
            }
            senão{
                prox = atual.pai;
            }
        }
        senão{
            prox = atual.pai;
        }
        ant = atual;
        atual = prox;
    }
}

procedimento de percurso Em Ordem que recebe uma arvore binaria e retorna uma lista encadeada decrescente
Registro NoListaEnc{
    inteiro chave;
    NoListaEnc proximo;
}
procedimento emOrdemParaLista(Arvore arv){
    cabeca = NIL;
    ultimo = NIL;
    ant = NIL;
    atual = arv.raiz;
    prox = NIL;
    enquanto atual != NIL{
        se ant == atual.pai{
            se atual.dir != NIL{
                prox = atual.dir;
            }
            senão{
                novo = NoListaEnc();
                novo.chave = atual.chave;
                novo.proximo = NIL;
                se ultimo == NIL{ cabeca = novo; }
                senão{ ultimo.proximo = novo; }
                ultimo = novo;

                se atual.esq != NIL{ 
                    prox = atual.esq; }
                senão{ 
                    prox = atual.pai; }
            }
        }
        senão se ant == atual.dir{
            novo = NoListaEnc();
            novo.chave = atual.chave;
            novo.proximo = NIL;
            se ultimo == NIL{ 
                cabeca = novo; }
            senão{ 
                ultimo.proximo = novo; }
            ultimo = novo;

            se atual.esq != NIL{ prox = atual.esq; }
            senão{ prox = atual.pai; }
        }
        senão{
            prox = atual.pai;
        }
        ant = atual;
        atual = prox;
    }
    retorne cabeca;
}
