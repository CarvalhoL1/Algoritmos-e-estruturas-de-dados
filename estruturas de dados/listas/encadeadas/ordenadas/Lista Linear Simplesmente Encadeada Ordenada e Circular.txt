registro NoListaSimplesEnc{
    inteiro chave;
    NoListaSimplesEnc proximo;
}
registro ListaSimplesEnc{
    NoListaSimplesEnc cabeca;
}
procedimento imprimirListaEnc(L){
    pt = L.cabeca;
    se pt != NIL{
        faça{
            imprimir(pt.chave);
            pt = pt.proximo;
        }enquanto pt != L.cabeca;}
}
procedimento busca(x, ant, L){ //Busca retorna o primeiro elemento >= a x, para inserção, x deve ser anterior a ele
    pt  = L.cabeca
    ant = NIL;
    se pt != NIL{
        se pt.chave >=x{
            ant = L.cabeca;
            enquanto ant.proximo != cabeca{
                ant = ant.proximo;
            }
            retorne pt;
        }
        senão{
        ant = pt;
        pt = pt.proximo;
        enquanto pt != L.cabeca {
        
            se(pt.chave >= x){
                retorne pt;
            }
            ant = pt;
            pt = pt.proximo;
            } 
        }
    }
    senão{
        imprimir("Lista vazia");
    }
    retorne NIL;
}

procedimento inserir(x, L){
    ant = NIL;
    pt = busca(x.chave, ant, L);
    se L.cabeca == NIL{ //Lista vazia
        L.cabeca = x;
        x.proximo = x;
    }
    senão se pt != x e pt != NIL {
        
        senão se pt == L.cabeca{ //x é o novo menor termo
            x.proximo = L.cabeca;
            L.cabeca = x;
            ant.proximo = L.cabeca;
        }
        senão{ //insere no meio normalmente, entre ant e pt
            ant.proximo = x;
            x.proximo = pt;
        }
        
    }
    senão se pt == NIL { //x maior de todos, caso pt seja nulo (não achou elemento menor)
            ant.proximo = x;
            x.proximo = L.cabeca;
        }
    senão {
            imprimir("Nó " + x.chave + " já existe");
        }
    
}

procedimento remover(x, L){
    ant = NIL;
    pt = busca(x, ant, L);
    se pt != NIL e pt.chave == x{
        se pt == L.cabeca{
            se pt.proximo != L.cabeca{
                L.cabeca = pt.proximo;
                ant.proximo = L.cabeca;
                } //O elemento a ser removido é a cabeça, então passa o status de cabeça pro proximo
            senão{
                L.cabeca = NIL; //só tem um elemento
            }
        }
        senão{
            ant.proximo = pt.proximo; //Removendo um nó que não é a cabeça
        }
        pt.proximo = NIL;
    }
    senão{
        imprimir("Nó não encontrado");
    }
    retorne pt;
}
