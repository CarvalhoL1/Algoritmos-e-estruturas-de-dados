procedimento heapsort(A, tamanhoA){
    tamanhoHeap = tamanhoA;
    construirMaxHeap(A, tamanhoHeap);
    para i = tamanhoA até 2 passo - 1{
        trocar(A[1], A[i]);
        tamanhoHeap = tamanhoHeap - 1;
        maxHeapfy(A, 1, tamanhoHeap);
    }
}

procedimento construirMaxHeap(A, tamanhoHeap){
    para i = floor(tamanhoHeap/2) até 1 passo - 1{
        maxHeapfy(A, i, tamanhoHeap);
    }
}

//i -> indice do nó em analise
procedimento maxHeapfy(A, i, tamanhoHeap){
    l = retornarIndiceFilhoEsquerda(i);
    r = retornarIndiceFilhoDireita(i);
    se (l <= tamanhoHeap) e (A[l] > A[i]){
        maior = l
    }
    senão{
        maior = i;
    }
    se (r <= tamanhoHeap) e (A[r] > A[maior]){
        maior = r;
    }
    se maior != i{
        trocar(A[i], A[maior]);
        maxHeapfy(A, maior, tamanhoHeap);
    }
}

procedimento retornarIndicePai(i){
    retorne floor(i / 2);}

procedimento retornarIndiceFilhoEsquerda(i){
    retorne i * 2;}

procedimento retornarIndiceFilhoDireita(i){
    retorne (i * 2) + 1;}

procedimento trocar(A[i], A[j]){
    temp = A[i]
    A[i] = A[j]
    A[j] = temp
}
