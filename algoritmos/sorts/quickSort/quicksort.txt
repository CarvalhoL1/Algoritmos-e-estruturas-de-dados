//A -> Array
//p -> indice da posição mais a esquerda
//r -> indice da posição mais a direita
procedimento quicksort(A, p, r){
    se p < r{
        q = particionar(A, p, r);
        quicksort(A, p, q-1);
        quicksort(A, q+1, r);
    }
}
procedimento particionar(A, p, r){
    x = A[r];
    i = p - 1;
    para j = p até r-1{
        se A[j] <= x{
            i = i + 1;
            trocar(A[i], A[j]);
        }
    }
    trocar(A[i+1], A[r]);
    retorne i+1;
}

procedimento trocar(A[i], A[j]){
    temp = A[i]
    A[i] = A[j]
    A[j] = temp
}
